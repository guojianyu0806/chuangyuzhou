+param([switch]$NoOpen)
+$ErrorActionPreference = 'Stop'
+$workbenchUrl = 'http://127.0.0.1:4173'
+$workbenchRoot = $PSScriptRoot
+
+function Test-WorkbenchReady {
+    try {
+        $health = Invoke-RestMethod -Uri "$workbenchUrl/api/health" -TimeoutSec 2
+        return ($health.version -eq '0.1.0' -and $health.backendConfigured -eq $false)
+    } catch { return $false }
+}
+
+try {
+    if (-not (Test-WorkbenchReady)) {
+        $nodeCommand = Get-Command node -ErrorAction SilentlyContinue
+        if (-not $nodeCommand) { throw 'Node.js was not found. Please install Node.js before starting the workbench.' }
+        $runtimeDirectory = Join-Path $workbenchRoot '.runtime'
+        New-Item -ItemType Directory -Path $runtimeDirectory -Force | Out-Null
+        $runStamp = Get-Date -Format 'yyyyMMdd-HHmmss-fff'
+        $outLog = Join-Path $runtimeDirectory "$runStamp.stdout.log"
+        $errLog = Join-Path $runtimeDirectory "$runStamp.stderr.log"
+        $previousHost = $env:HOST
+        $previousPort = $env:PORT
+        try {
+            $env:HOST = '127.0.0.1'
+            $env:PORT = '4173'
+            $workbenchProcess = Start-Process -FilePath $nodeCommand.Source -ArgumentList 'server.mjs' -WorkingDirectory $workbenchRoot -WindowStyle Hidden -RedirectStandardOutput $outLog -RedirectStandardError $errLog -PassThru
+        } finally {
+            $env:HOST = $previousHost
+            $env:PORT = $previousPort
+        }
+        $ready = $false
+        for ($attempt = 0; $attempt -lt 30; $attempt++) {
+            if (Test-WorkbenchReady) { $ready = $true; break }
+            $workbenchProcess.Refresh()
+            if ($workbenchProcess.HasExited) { break }
+            Start-Sleep -Milliseconds 150
+        }
+        if (-not $ready) {
+            $detail = if (Test-Path -LiteralPath $errLog) { Get-Content -LiteralPath $errLog -Raw } else { '' }
+            throw "Workbench did not become ready. See $errLog. $detail"
+        }
+    }
+    Write-Output "Workbench is ready: $workbenchUrl"
+    Write-Output 'The server runs in the background. Closing this launcher will not stop it.'
+    if (-not $NoOpen) { Start-Process $workbenchUrl }
+} catch {
+    Write-Error $_ -ErrorAction Continue
+    exit 1
+}
